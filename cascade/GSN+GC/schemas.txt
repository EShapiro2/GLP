Reflexive predicates: knows

establish: identity(Alice) <-
    not identity(Alice) guarded by Alice

acquaint: knows(Alice, Bob), knows(Bob, Alice) <-
    identity(Alice), identity(Bob) guarded by Alice, Bob

introduce: knows(Bob, Carol) <-
    identity(Bob), knows(Bob, Alice), identity(Alice), knows(Alice, Bob), knows(Alice, Carol) guarded by Bob, Alice

befriend: friend(Alice, Bob), friend(Bob, Alice) <-
    identity(Alice), knows(Alice, Bob), not friend(Alice, Bob), identity(Bob), knows(Bob, Alice), not friend(Bob, Alice) guarded by Alice, Bob

unfriend: not friend(Alice, Bob), not member(Alice, _, Bob), not listed(Alice, Bob, _), not friend(Bob, Alice), not member(Bob, _, Alice), not listed(Bob, Alice, _) <-
    identity(Alice), friend(Alice, Bob), identity(Bob), friend(Bob, Alice) guarded by Alice

message: message(Bob, Alice, Text) <-
    identity(Bob), friend(Bob, Alice), identity(Alice), friend(Alice, Bob) guarded by Alice

create: leader(Alice, Grupo), member(Alice, Grupo, Alice), listed(Alice, Alice, Grupo) <-
    identity(Alice), not leader(Alice, Grupo) guarded by Alice

close: not leader(Alice, Grupo), not listed(Alice, _, Grupo), forall Bob (listed(Alice, Bob, Grupo) -> not member(Bob, Grupo, Alice)) <-
    identity(Alice), leader(Alice, Grupo), forall Bob (listed(Alice, Bob, Grupo) -> identity(Bob), member(Bob, Grupo, Alice)) guarded by Alice

join: member(Bob, Grupo, Alice), listed(Alice, Bob, Grupo) <-
    identity(Bob), friend(Bob, Alice), not member(Bob, Grupo, _), identity(Alice), leader(Alice, Grupo), friend(Alice, Bob) guarded by Bob, Alice

remove: not member(Bob, Grupo, Alice), not listed(Alice, Bob, Grupo) <-
    identity(Bob), member(Bob, Grupo, Alice), identity(Alice), leader(Alice, Grupo), listed(Alice, Bob, Grupo) guarded by Alice

leave: not member(Bob, Grupo, Alice), not listed(Alice, Bob, Grupo) <-
    identity(Bob), member(Bob, Grupo, Alice), identity(Alice), listed(Alice, Bob, Grupo) guarded by Bob

post: posted(Alice, Text, Bob, Grupo) <-
    identity(Alice), not posted(Alice, Text, Bob, Grupo), identity(Bob), member(Bob, Grupo, Alice) guarded by Bob

deliver: group_delivered(Alice, Bob, Text, Grupo, Carol), group_message(Carol, Bob, Text, Grupo, Alice) <-
    identity(Alice), leader(Alice, Grupo), posted(Alice, Text, Bob, Grupo), listed(Alice, Carol, Grupo), not group_delivered(Alice, Bob, Text, Grupo, Carol), identity(Carol)

    identity(Alice), leader(Alice, Grupo), posted(Alice, Text, Bob, Grupo), not group_delivered(Alice, Bob, Text, Grupo, Alice)

issue: Sum.coin(Alice, Alice) <-
    identity(Alice) guarded by Alice

    identity(Alice), knows(Alice, Bob), Sum.coin(Alice, Carol), identity(Bob), knows(Bob, Alice), Total.coin(Bob, Dave) guarded by Alice, Bob

    identity(Alice), knows(Alice, Bob), Sum.coin(Alice, Bob), identity(Bob) guarded by Alice
