establish(Alice?) :
    at Alice: +identity

meet(Alice?, Bob?) :
    at Alice: identity, +knows(Bob)
    at Bob: identity, +knows(Alice)

introduce(Alice?, Bob?; c) :
    at Alice: identity, knows(Bob), knows(c)
    at Bob: identity, knows(Alice), +knows(c)

mint(Alice?) :
    at Alice: identity, +coin(Alice)

swap(Alice?, Bob?; u, v) :
    at Alice: identity, knows(Bob), +coin(v), -coin(u)
    at Bob: identity, knows(Alice), +coin(u), -coin(v)

pay(Alice?, Bob) :
    at Alice: identity, knows(Bob), -coin(Bob)
    at Bob: identity, +coin(Bob)

redeem(Alice?, Bob; r) :
    at Alice: identity, knows(Bob), +coin(r), -coin(Bob)
    at Bob: identity, +coin(Bob), -coin(r)
