establish(Alice?) :
    at Alice: not identity, +identity

acquaint(Alice?, Bob?) :
    at Alice: identity, +knows(Bob)
    at Bob: identity, +knows(Alice)

introduce(Alice?, Bob?; c) :
    at Alice: identity, knows(Bob), knows(c)
    at Bob: identity, knows(Alice), +knows(c)

mint(Alice?; k) :
    at Alice: identity, +k.coin(Alice)

swap(Alice?, Bob?; u, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.coin(v), +knows(v), -k.coin(u)
    at Bob: identity, knows(Alice), +k.coin(u), +knows(u), -k1.coin(v)

pay(Alice?, Bob; k) :
    at Alice: identity, knows(Bob), -k.coin(Bob)
    at Bob: identity, +k.coin(Bob)

redeem(Alice?, Bob; r) :
    at Alice: identity, knows(Bob), +coin(r), +knows(r), -coin(Bob)
    at Bob: identity, +coin(Bob), -coin(r)

dmint(Alice?; f, k) :
    at Alice: identity, +k.dcoin(Alice, f)

swap_dc(Alice?, Bob?; f, u, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.coin(v), +knows(v), -k.dcoin(u, f)
    at Bob: identity, knows(Alice), +k.dcoin(u, f), +knows(u), -k1.coin(v)

swap_cd(Alice?, Bob?; f, u, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.dcoin(v, f), +knows(v), -k.coin(u)
    at Bob: identity, knows(Alice), +k.coin(u), +knows(u), -k1.dcoin(v, f)

swap_dd(Alice?, Bob?; f, f1, u, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.dcoin(v, f1), +knows(v), -k.dcoin(u, f)
    at Bob: identity, knows(Alice), +k.dcoin(u, f), +knows(u), -k1.dcoin(v, f1)

pay_d(Alice?, Bob; f, k) :
    at Alice: identity, knows(Bob), -k.dcoin(Bob, f)
    at Bob: identity, +k.dcoin(Bob, f)

redeem_dc(Alice?, Bob; f, r) :
    at Alice: identity, knows(Bob), +coin(r), +knows(r), -dcoin(Bob, f)
    at Bob: identity, +dcoin(Bob, f), -coin(r)

redeem_cd(Alice?, Bob; f, r) :
    at Alice: identity, knows(Bob), +dcoin(r, f), +knows(r), -coin(Bob)
    at Bob: identity, +coin(Bob), -dcoin(r, f)

redeem_dd(Alice?, Bob; f, f1, r) :
    at Alice: identity, knows(Bob), +dcoin(r, f1), +knows(r), -dcoin(Bob, f)
    at Bob: identity, +dcoin(Bob, f), -dcoin(r, f1)
