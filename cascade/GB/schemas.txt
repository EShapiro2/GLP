establish(Alice?) :
    at Alice: not identity, +identity

meet(Alice?, Bob?) :
    at Alice: identity, +knows(Bob)
    at Bob: identity, +knows(Alice)

introduce(Alice?, Bob?; c) :
    at Alice: identity, knows(Bob), knows(c)
    at Bob: identity, knows(Alice), +knows(c)

mint_coin(Alice?; k) :
    at Alice: identity, +k.coin(Alice)

mint_bond(Alice?; d, k, t) | d > t :
    at Alice: identity, date(t), +k.bond(Alice, d)

first_date(Alice; d) :
    at Alice: identity, today(d), not dated, +date(d), +dated

advance(Alice; t, t1, d) | t1 = t + 1, t1 <= d :
    at Alice: identity, today(d), +date(t1), -date(t)

swap_cc(Alice?, Bob?; u, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.coin(v), +knows(v), -k.coin(u)
    at Bob: identity, knows(Alice), +k.coin(u), +knows(u), -k1.coin(v)

swap_cb(Alice?, Bob?; u, k, v, d1, k1) :
    at Alice: identity, knows(Bob), +k1.bond(v, d1), +knows(v), -k.coin(u)
    at Bob: identity, knows(Alice), +k.coin(u), +knows(u), -k1.bond(v, d1)

swap_bc(Alice?, Bob?; u, d, k, v, k1) :
    at Alice: identity, knows(Bob), +k1.coin(v), +knows(v), -k.bond(u, d)
    at Bob: identity, knows(Alice), +k.bond(u, d), +knows(u), -k1.coin(v)

swap_bb(Alice?, Bob?; u, d, k, v, d1, k1) :
    at Alice: identity, knows(Bob), +k1.bond(v, d1), +knows(v), -k.bond(u, d)
    at Bob: identity, knows(Alice), +k.bond(u, d), +knows(u), -k1.bond(v, d1)

pay(Alice?, Bob; k) :
    at Alice: identity, knows(Bob), -k.coin(Bob)
    at Bob: identity, +k.coin(Bob)

redeem_cc(Alice?, Bob; r) :
    at Alice: identity, knows(Bob), +coin(r), +knows(r), -coin(Bob)
    at Bob: identity, +coin(Bob), -coin(r)

redeem_cb(Alice?, Bob; r, d1) :
    at Alice: identity, knows(Bob), +bond(r, d1), +knows(r), -coin(Bob)
    at Bob: identity, +coin(Bob), -bond(r, d1)

redeem_bc(Alice?, Bob; d, r, t) | d <= t :
    at Alice: identity, knows(Bob), +coin(r), +knows(r), -bond(Bob, d)
    at Bob: identity, date(t), +bond(Bob, d), -coin(r)

redeem_bb(Alice?, Bob; d, r, d1, t) | d <= t :
    at Alice: identity, knows(Bob), +bond(r, d1), +knows(r), -bond(Bob, d)
    at Bob: identity, date(t), +bond(Bob, d), -bond(r, d1)

deposit_c(Alice?, Carol?; u, k) :
    at Alice: identity, knows(Carol), -k.coin(u)
    at Carol: identity, knows(Alice), +k.coin(u), +knows(u)

deposit_b(Alice?, Carol?; u, d, k) :
    at Alice: identity, knows(Carol), -k.bond(u, d)
    at Carol: identity, knows(Alice), +k.bond(u, d), +knows(u)

release_c(Carol?, Bob?; u, k) :
    at Carol: identity, knows(Bob), -k.coin(u)
    at Bob: identity, knows(Carol), +k.coin(u), +knows(u)

release_b(Carol?, Bob?; u, d, k) :
    at Carol: identity, knows(Bob), -k.bond(u, d)
    at Bob: identity, knows(Carol), +k.bond(u, d), +knows(u)

return_c(Carol?, Alice?; u, k) :
    at Carol: identity, knows(Alice), -k.coin(u)
    at Alice: identity, knows(Carol), +k.coin(u), +knows(u)

return_b(Carol?, Alice?; u, d, k) :
    at Carol: identity, knows(Alice), -k.bond(u, d)
    at Alice: identity, knows(Carol), +k.bond(u, d), +knows(u)
