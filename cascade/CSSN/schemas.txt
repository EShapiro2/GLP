Reflexive predicates: friend, knows

establish(Alice?) :
    at Alice: not identity, not retired, +identity

meet(Alice?, Bob?) :
    at Alice: identity, not child, not retired, +knows(Bob)
    at Bob: identity, not child, not retired, +knows(Alice)

introduce(Alice?, Bob?; c) :
    at Alice: identity, knows(Bob), knows(c), not retired
    at Bob: identity, knows(Alice), not child, not retired, +knows(c)

befriend(Alice?, Bob?) :
    at Alice: identity, knows(Bob), not child, not friend(Bob), not retired, +friend(Bob)
    at Bob: identity, knows(Alice), not child, not friend(Alice), not retired, +friend(Alice)

unfriend(Alice?, Bob) :
    at Alice: identity, not creator(Bob), not parenting(Bob), not retired, -friend(Bob)
    at Bob: identity, -friend(Alice)

message(Alice?, Bob; x) :
    at Alice: identity, not retired
    at Bob: identity, friend(Alice), +received(x, Alice)

create(Alice?; n) :
    at Alice: identity, not member(<Alice>.n), not closed(n), not retired, +member(<Alice>.n), +listed(Alice, n)

join(Alice?, Bob?; n) :
    at Alice: identity, member(<Alice>.n), friend(Bob), not child, not retired, +listed(Bob, n)
    at Bob: identity, friend(Alice), not child, not member(<Alice>.n), not retired, +member(<Alice>.n)

leave(Alice, Bob?; n) :
    at Alice: identity, -listed(Bob, n)
    at Bob: identity, not retired, -member(<Alice>.n)

close(Alice?, Alice[listed(*, n)]; n) :
    at Alice: identity, not retired, +closed(n), -listed(Alice, n)
    at Alice[listed(*, n)]: identity, -member(<Alice>.n)

drop(Alice, Bob; n) :
    at Alice: identity, not friend(Bob), -listed(Bob, n)
    at Bob: identity, -member(<Alice>.n)

post_own(Alice?; x, n) :
    at Alice: identity, member(<Alice>.n), not posted(x, n), not retired, +posted(x, n)

post(Alice?, Bob; x, n) :
    at Alice: identity, member(<Bob>.n), not retired
    at Bob: identity, member(<Bob>.n), listed(Alice, n), friend(Alice), not posted(x, n), +posted(x, n)

deliver(Alice, Bob; x, n) :
    at Alice: identity, member(<Alice>.n), posted(x, n), listed(Bob, n), friend(Bob), not delivered(x, Bob, n), +delivered(x, Bob, n)
    at Bob: identity, member(<Alice>.n), +received_in(x, Alice, n)

enter(Alice?, Bob?) :
    at Alice: identity, not child, not retired, +created(Bob), +parenting(Bob), +friend(Bob), +knows(Bob)
    at Bob: not identity, not retired, +identity, +child, +creator(Alice), +friend(Alice), +knows(Alice)

make_parent(Alice?, Bob?, Carol?) :
    at Alice: identity, created(Carol), knows(Bob), not child, not made(Bob, Carol), not retired, +made(Bob, Carol)
    at Bob: identity, knows(Alice), not child, not parenting(Carol), not retired, +parenting(Carol), +knows(Carol)
    at Carol: identity, child, creator(Alice), not retired, +knows(Bob)

end_parent(Alice?, Bob; r) :
    at Alice: identity, created(r), not retired, -made(Bob, r)
    at Bob: identity, not created(r), -parenting(r)

of_age(Alice?, Bob, Alice[made(*, Bob)]) :
    at Alice: identity, not retired, -created(Bob), -parenting(Bob)
    at Bob: identity, -child, -creator(Alice)
    at Alice[made(*, Bob)]: identity, not created(Bob), -parenting(Bob)

retire(Alice?, Bob, Alice[made(*, Bob)]) :
    at Alice: identity, not retired, -created(Bob), -parenting(Bob)
    at Bob: identity, +retired, -child, -creator(Alice)
    at Alice[made(*, Bob)]: identity, not created(Bob), -parenting(Bob)

child_meet(Alice?, Bob?, Carol?) :
    at Alice: identity, child, not retired, +knows(Bob)
    at Bob: identity, not child, not retired, +knows(Alice)
    at Carol: identity, parenting(Alice), knows(Bob), not child, not retired

child_meet_2(Alice?, Bob?, Carol?, Dave?) :
    at Alice: identity, child, not retired, +knows(Bob)
    at Bob: identity, child, not retired, +knows(Alice)
    at Carol: identity, parenting(Alice), knows(Dave), not child, not retired
    at Dave: identity, parenting(Bob), knows(Carol), not child, not retired

child_befriend(Alice?, Bob?, Carol?, Dave?) :
    at Alice: identity, child, knows(Bob), not friend(Bob), not retired, +friend(Bob)
    at Bob: identity, child, knows(Alice), not friend(Alice), not retired, +friend(Alice)
    at Carol: identity, parenting(Alice), friend(Dave), not child, not retired
    at Dave: identity, parenting(Bob), friend(Carol), not child, not retired

child_adult_befriend(Alice?, Bob?, Carol?) :
    at Alice: identity, child, knows(Bob), not friend(Bob), not retired, +friend(Bob)
    at Bob: identity, knows(Alice), not child, not friend(Alice), not retired, +friend(Alice)
    at Carol: identity, parenting(Alice), friend(Bob), not child, not retired

child_unfriend(Alice, Bob, Carol?) :
    at Alice: identity, child, -friend(Bob)
    at Bob: identity, not parenting(Alice), -friend(Alice)
    at Carol: identity, parenting(Alice), not child, not retired

child_join(Alice?, Bob?, Carol?; n) :
    at Alice: identity, member(<Alice>.n), friend(Bob), not child, not retired, +listed(Bob, n)
    at Bob: identity, child, friend(Alice), not member(<Alice>.n), not retired, +member(<Alice>.n)
    at Carol: identity, parenting(Bob), not child, not retired

child_join_2(Alice?, Bob?, Carol?, Dave?; n) :
    at Alice: identity, child, member(<Alice>.n), friend(Bob), not retired, +listed(Bob, n)
    at Bob: identity, child, friend(Alice), not member(<Alice>.n), not retired, +member(<Alice>.n)
    at Carol: identity, parenting(Alice), not child, not retired
    at Dave: identity, parenting(Bob), not child, not retired

child_leave(Alice, Bob, Carol?; n) :
    at Alice: identity, -listed(Bob, n)
    at Bob: identity, child, -member(<Alice>.n)
    at Carol: identity, parenting(Bob), not child, not retired

child_close(Alice, Bob?, Alice[listed(*, n)]; n) :
    at Alice: identity, child, +closed(n), -listed(Alice, n)
    at Bob: identity, parenting(Alice), not child, not retired
    at Alice[listed(*, n)]: identity, -member(<Alice>.n)
