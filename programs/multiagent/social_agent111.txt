%% social_agent.glp - Social graph agent with friends list
%% Based on GLP-2025 paper architecture
%% For use with glp_multiagent Flutter app
%% 
%% Supports multiple pre-configured friends via FriendPairs list.
%% Supports friend-mediated introduction protocol.

%% ========== Type Definitions ==========

%% Network messages between agents
NetMsg ::= msg(_, _, _).

%% User commands (input from user)
UserCmd ::= send(_, _) ; introduce(_, _) ; accept_intro(_, _) ; reject_intro(_, _).

%% User feedback messages (output to user)
UserFeedback ::= sent(_, _) ; received(_, _) ; received_ping(_) ; received_pong(_) 
              ; intro_offer(_, _, _) ; introduced(_, _) ; accepted_intro(_) ; rejected_intro(_)
              ; error(_, _).

%% All message types in streams
Msg ::= NetMsg ; UserCmd ; UserFeedback.

%% Message streams
MsgStream ::= [] ; [Msg | MsgStream].

%% Channel pair: reader for input, writer for output
Channel ::= ch(MsgStream?, MsgStream).

%% Friend list entry: (name, output stream)
FriendEntry ::= (_, MsgStream).
FriendList ::= [] ; [FriendEntry | FriendList].

%% Friend pairs for initialization (name, output stream)
FriendPairs ::= [] ; [FriendEntry | FriendPairs].

%% Lookup status
Status ::= found ; not_found.

%% ========== Procedure Declarations ==========

procedure agent(_?, FriendPairs?, Channel?, Channel?).
procedure build_friends(FriendPairs?, MsgStream, MsgStream, FriendList).
procedure social_graph(_?, MsgStream?, FriendList?).
procedure lookup_send(_?, Msg?, FriendList?, FriendList, Status).
procedure handle_send_status(Status?, _?, _?, FriendList?, FriendList).
procedure handle_intro_status(Status?, Status?, _?, _?, FriendList?, FriendList).
procedure handle_ping_status(Status?, _?, _?, FriendList?, FriendList).
procedure add_friend(_?, MsgStream?, FriendList?, FriendList).
procedure tag_stream(_?, MsgStream?, MsgStream).
procedure append(FriendList?, FriendList?, FriendList).
procedure merge(MsgStream?, MsgStream?, MsgStream).

%% ========== Agent Initialization ==========

%% agent(Id, FriendPairs, UserCh, NetCh)
%% FriendPairs is a list of (Name, OutStream) pairs for pre-configured friends
agent(Id, FriendPairs, ch(UserIn, UserOut?), ch(NetIn, NetOut?)) :-
    merge(UserIn?, NetIn?, In),
    build_friends(FriendPairs?, UserOut, NetOut, Fs),
    social_graph(Id?, In?, Fs?).

%% Build friends list from pairs, adding user and net channels
build_friends(FriendPairs, UserOut, NetOut, Fs?) :-
    append([(user, UserOut?), (net, NetOut?)], FriendPairs?, Fs).

%% ========== Social Graph Main Loop ==========

%% User sends message to a friend
social_graph(Id, [send(To, Content)|In], Fs) :-
    ground(Id?), ground(To?), ground(Content?) |
    lookup_send(To?, msg(Id?, To?, Content?), Fs?, Fs1, Status),
    handle_send_status(Status?, To?, Content?, Fs1?, Fs2),
    social_graph(Id?, In?, Fs2?).

%% ========== Friend-Mediated Introduction Protocol ==========

%% User requests introduction: introduce(P, Q)
%% Bob types: introduce(alice, charlie)
%% Bob creates a fresh channel pair and sends half to each party
social_graph(Id, [introduce(P, Q)|In], Fs) :-
    ground(Id?), ground(P?), ground(Q?) |
    %% Create fresh channel pair: PtoQ and QtoP are the two halves
    %% P gets ch(QtoP?, PtoQ) - reads from Q, writes to Q
    %% Q gets ch(PtoQ?, QtoP) - reads from P, writes to P
    lookup_send(P?, msg(Id?, P?, intro(Q?, ch(QtoP?, PtoQ))), Fs?, Fs1, Status1),
    lookup_send(Q?, msg(Id?, Q?, intro(P?, ch(PtoQ?, QtoP))), Fs1?, Fs2, Status2),
    handle_intro_status(Status1?, Status2?, P?, Q?, Fs2?, Fs3),
    social_graph(Id?, In?, Fs3?).

%% Handle introduction status
handle_intro_status(found, found, P, Q, Fs, Fs1?) :-
    lookup_send(user, introduced(P?, Q?), Fs?, Fs1, _).

handle_intro_status(not_found, _, P, _, Fs, Fs1?) :-
    lookup_send(user, error(intro_failed, P?), Fs?, Fs1, _).

handle_intro_status(_, not_found, _, Q, Fs, Fs1?) :-
    lookup_send(user, error(intro_failed, Q?), Fs?, Fs1, _).

%% Receive introduction from a friend
%% Alice receives: msg(bob, alice, intro(charlie, ch(CharlieToAlice?, AliceToCharlie)))
social_graph(Id, [msg(From, Id?, intro(Other, Ch))|In], Fs) :-
    ground(Id?), ground(From?), ground(Other?) |
    %% Notify user about the introduction offer
    lookup_send(user, intro_offer(From?, Other?, Ch?), Fs?, Fs1, _),
    social_graph(Id?, In?, Fs1?).

%% User accepts introduction: accept_intro(Other, Ch)
%% Alice types: accept_intro(charlie, Ch) where Ch was shown in intro_offer
social_graph(Id, [accept_intro(Other, ch(FIn, FOut))|In], Fs) :-
    ground(Id?), ground(Other?) |
    %% Tag messages from this new friend
    tag_stream(Other?, FIn?, Tagged),
    %% Merge the new friend's input into our main input
    merge(In?, Tagged?, In1),
    %% Add friend to our friends list
    add_friend(Other?, FOut?, Fs?, Fs1),
    lookup_send(user, accepted_intro(Other?), Fs1?, Fs2, _),
    social_graph(Id?, In1?, Fs2?).

%% User rejects introduction
social_graph(Id, [reject_intro(Other, _)|In], Fs) :-
    ground(Id?), ground(Other?) |
    lookup_send(user, rejected_intro(Other?), Fs?, Fs1, _),
    social_graph(Id?, In?, Fs1?).

%% ========== Ping/Pong Protocol ==========

%% Receive ping - reply with pong
social_graph(Id, [msg(From, Id?, ping)|In], Fs) :-
    ground(Id?), ground(From?) |
    lookup_send(user, received_ping(From?), Fs?, Fs1, _),
    lookup_send(From?, msg(Id?, From?, pong), Fs1?, Fs2, Status),
    handle_ping_status(Status?, Id?, From?, Fs2?, Fs3),
    social_graph(Id?, In?, Fs3?).

%% Receive pong - display only
social_graph(Id, [msg(From, Id?, pong)|In], Fs) :-
    ground(Id?), ground(From?) |
    lookup_send(user, received_pong(From?), Fs?, Fs1, _),
    social_graph(Id?, In?, Fs1?).

%% Receive other message - display only
social_graph(Id, [msg(From, Id?, Content)|In], Fs) :-
    ground(Id?), ground(From?), ground(Content?) |
    lookup_send(user, received(From?, Content?), Fs?, Fs1, _),
    social_graph(Id?, In?, Fs1?).

%% Unknown command - report error
social_graph(Id, [Cmd|In], Fs) :-
    ground(Id?), ground(Cmd?), otherwise |
    lookup_send(user, error(unknown_command, Cmd?), Fs?, Fs1, _),
    social_graph(Id?, In?, Fs1?).

%% Empty input - done
social_graph(_, [], _).

%% ========== Status Handlers ==========

handle_send_status(found, To, Content, Fs, Fs1?) :-
    lookup_send(user, sent(To?, Content?), Fs?, Fs1, _).

handle_send_status(not_found, To, _, Fs, Fs1?) :-
    lookup_send(user, error(unknown_recipient, To?), Fs?, Fs1, _).

handle_ping_status(found, _, _, Fs, Fs?).

handle_ping_status(not_found, _, From, Fs, Fs1?) :-
    lookup_send(user, error(ping_reply_failed, From?), Fs?, Fs1, _).

%% ========== Friend Management ==========

%% Add a new friend to the friends list
add_friend(Name, Out, Fs, [(Name?, Out?)|Fs?]).

%% Tag messages from a stream with the sender's name
%% Transforms raw stream into msg(From, _, Content) format
tag_stream(_, [], []).
tag_stream(From, [Content|Rest], [msg(From?, _, Content?)|Tagged?]) :-
    ground(From?) |
    tag_stream(From?, Rest?, Tagged).

%% ========== Lookup Send ==========

lookup_send(_, _, [], [], not_found).

lookup_send(Key, Msg, [(K, Out)|Rest], [(K?, Out1?)|Rest?], found) :-
    Key? =?= K? |
    Out? = [Msg?|Out1].

lookup_send(Key, Msg, [Pair|Rest], [Pair?|Rest1?], Status?) :-
    otherwise |
    lookup_send(Key?, Msg?, Rest?, Rest1, Status).

%% ========== Append ==========

append([], Ys, Ys?).
append([X|Xs], Ys, [X?|Zs?]) :- append(Xs?, Ys?, Zs).

%% ========== Merge ==========

merge([H|T1], T2, [H?|T3?]) :- merge(T1?, T2?, T3).
merge(T1, [H|T2], [H?|T3?]) :- merge(T1?, T2?, T3).
merge([], Ys, Ys?).
merge(Xs, [], Xs?).
